# Эмулятор почтового шлюза в кластере. Состояния нет, поэтому Deployment; Service и PDB — в service.yaml и pdb.yaml.
apiVersion: apps/v1
kind: Deployment
metadata:
  name: mailgw
  labels:
    app.kubernetes.io/name: mailgw
    {{- include "soldout.labels" . | nindent 4 }}
spec:
  replicas: {{ $m.replicas }}
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxUnavailable: 0
      maxSurge: 1
  selector:
    matchLabels:
      app.kubernetes.io/name: mailgw
  template:
    metadata:
      labels:
        app.kubernetes.io/name: mailgw
        {{- include "soldout.labels" . | nindent 8 }}
    spec:
      terminationGracePeriodSeconds: {{ $m.terminationGracePeriodSeconds }}
      securityContext:
        runAsNonRoot: true
        runAsUser: 65532
        seccompProfile:
          type: RuntimeDefault
      containers:
        - name: mailgw
          image: "{{ $m.image }}:{{ required "image.tag обязателен" .Values.image.tag }}"
          imagePullPolicy: {{ .Values.image.pullPolicy }}
          command: ["/mailgw"]
          ports:
            - name: http
              containerPort: {{ $m.port }}
          env:
            - name: MAILGW_ADDR
              value: ":{{ $m.port }}"
            - name: OTEL_EXPORTER_OTLP_ENDPOINT
              value: {{ .Values.external.otlpEndpoint | quote }}
            - name: GOMEMLIMIT
              value: {{ $m.gomemlimit | quote }}
            {{- range $k, $v := $m.env }}
            - name: {{ $k }}
              value: {{ $v | quote }}
            {{- end }}
          resources:
            {{- toYaml $m.resources | nindent 12 }}
          readinessProbe:
            httpGet: { path: {{ $m.probes.readiness.path }}, port: http }
            periodSeconds: {{ $m.probes.readiness.periodSeconds }}
            failureThreshold: {{ $m.probes.readiness.failureThreshold }}
          livenessProbe:
            httpGet: { path: {{ $m.probes.liveness.path }}, port: http }
            periodSeconds: {{ $m.probes.liveness.periodSeconds }}
            failureThreshold: {{ $m.probes.liveness.failureThreshold }}
          lifecycle:
            preStop:
              sleep:
                seconds: {{ $m.preStopSleepSeconds }}
          securityContext:
            {{- include "soldout.containerSecurity" . | nindent 12 }}
{{- end }}
